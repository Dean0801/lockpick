```rust
pub mod types;

use std::path::Path;
use types::LockpickConfig;

/// Load .lockpickrc.json or .lockpickrc.yaml from project root.
/// Returns Default if no config file found.
pub fn load_config(project_root: &Path) -> LockpickConfig {
    let json_path = project_root.join(".lockpickrc.json");
    if json_path.exists() {
        if let Ok(content) = std::fs::read_to_string(&json_path) {
            if let Ok(config) = serde_json::from_str::<LockpickConfig>(&content) {
                return config;
            }
        }
    }

    let yaml_path = project_root.join(".lockpickrc.yaml");
    if yaml_path.exists() {
        if let Ok(content) = std::fs::read_to_string(&yaml_path) {
            if let Ok(config) = serde_yaml::from_str::<LockpickConfig>(&content) {
                return config;
            }
        }
    }

    LockpickConfig::default()
}
```

```rust
/// Commands to skip (not real packages, just runners)
const SKIP_COMMANDS: &[&str] = &[
    "node", "npx", "npm", "yarn", "pnpm", "run", "exec",
    "env", "sh", "bash", "echo", "cd", "rm", "cp", "mkdir",
    "cat", "exit", "true", "false", "test",
];

/// Extract the first meaningful CLI command from a script value.
/// e.g. "cross-env NODE_ENV=prod node dist/index.js" → "cross-env"
/// e.g. "npx eslint ." → "eslint" (skip npx, take next token)
fn extract_command(script_value: &str) -> Option<String> {
    let tokens: Vec<&str> = script_value.split_whitespace().collect();
    if tokens.is_empty() {
        return None;
    }

    // Walk tokens, skip runners like npx/npm/yarn/node
    for token in &tokens {
        let cmd = token.trim();
        // Skip environment variable assignments (KEY=VALUE)
        if cmd.contains('=') && !cmd.starts_with('-') {
            continue;
        }
        // Skip flags
        if cmd.starts_with('-') {
            continue;
        }
        // Skip known runners
        if SKIP_COMMANDS.contains(&cmd) {
            continue;
        }
        // Skip "run" subcommands like "npm run build"
        if cmd == "run" {
            continue;
        }
        return Some(cmd.to_string());
    }

    None
}

/// Extract package names used in package.json scripts
pub fn extract_script_deps(project_root: &Path) -> HashSet<String> {
    let pkg_path = project_root.join("package.json");
    let mut deps = HashSet::new();

    let content = match std::fs::read_to_string(&pkg_path) {
        Ok(c) => c,
        Err(_) => return deps,
    };

    let json: serde_json::Value = match serde_json::from_str(&content) {
        Ok(v) => v,
        Err(_) => return deps,
    };

    let scripts = match json.get("scripts").and_then(|s| s.as_object()) {
        Some(s) => s,
        None => return deps,
    };

    let alias_map = cli_to_package_map();

    for (_key, value) in scripts {
        if let Some(script_str) = value.as_str() {
            if let Some(cmd) = extract_command(script_str) {
                // Look up alias map first, fallback to command name itself
                let pkg_name = alias_map
                    .get(cmd.as_str())
                    .map(|s| s.to_string())
                    .unwrap_or(cmd);
                deps.insert(pkg_name);
            }
        }
    }

    deps
}
```

```rust
    "tsup.config.js",
];

/// Scan JS/TS config files in project root, return imported package names
pub fn extract_js_config_deps(project_root: &Path) -> HashSet<String> {
    let mut deps = HashSet::new();

    for filename in JS_CONFIG_FILES {
        let path = project_root.join(filename);
        if path.exists() {
            if let Ok(source) = std::fs::read_to_string(&path) {
                let imports = extract_imports_from_source(&source, &path);
                deps.extend(imports);
            }
        }
    }

    deps
}
```

```rust
/// Expand a short plugin name to possible full package names.
/// e.g. for ESLint: "react" → ["eslint-plugin-react", "eslint-config-react", "react"]
/// If the name already starts with a known prefix or is scoped (@), return as-is.
fn expand_plugin_name(short_name: &str, prefixes: &[&str]) -> Vec<String> {
    // Already a scoped package or already has a known prefix → return as-is
    if short_name.starts_with('@') || prefixes.iter().any(|p| short_name.starts_with(p)) {
        return vec![short_name.to_string()];
    }

    if prefixes.is_empty() {
        return vec![short_name.to_string()];
    }

    let mut candidates = Vec::new();
    for prefix in prefixes {
        candidates.push(format!("{prefix}{short_name}"));
    }
    // Also keep the original name as fallback
    candidates.push(short_name.to_string());
    candidates
}

/// Extract string values from a JSON array or string field
fn extract_json_string_values(value: &serde_json::Value) -> Vec<String> {
    match value {
        serde_json::Value::String(s) => vec![s.clone()],
        serde_json::Value::Array(arr) => arr
            .iter()
            .filter_map(|v| {
                match v {
                    serde_json::Value::String(s) => Some(s.clone()),
                    // Babel allows [["preset-name", { options }]]
                    serde_json::Value::Array(inner) => {
                        inner.first().and_then(|f| f.as_str().map(String::from))
                    }
                    _ => None,
                }
            })
            .collect(),
        // PostCSS plugins as object keys: { "autoprefixer": {}, "tailwindcss": {} }
        serde_json::Value::Object(obj) => obj.keys().cloned().collect(),
        _ => vec![],
    }
}
```

```rust
/// Scan JSON/YAML config files in project root, return referenced package names
pub fn extract_json_config_deps(project_root: &Path) -> HashSet<String> {
    let mut deps = HashSet::new();

    for rule in JSON_CONFIG_RULES {
        for filename in rule.filenames {
            let path = project_root.join(filename);
            if !path.exists() {
                continue;
            }

            let content = match std::fs::read_to_string(&path) {
                Ok(c) => c,
                Err(_) => continue,
            };

            // Parse as JSON or YAML → serde_json::Value
            let json: serde_json::Value = if filename.ends_with(".yml")
                || filename.ends_with(".yaml")
            {
                match serde_yaml::from_str(&content) {
                    Ok(v) => v,
                    Err(_) => continue,
                }
            } else {
                match serde_json::from_str(&content) {
                    Ok(v) => v,
                    Err(_) => continue,
                }
            };

            for field in rule.fields {
                if let Some(value) = json.get(*field) {
                    let names = extract_json_string_values(value);
                    for name in names {
                        let expanded = expand_plugin_name(&name, rule.prefixes);
                        deps.extend(expanded);
                    }
                }
            }
        }
    }

    deps
}

/// Combined: scan all config files (JS/TS + JSON/YAML)
pub fn extract_config_deps(project_root: &Path) -> HashSet<String> {
    let mut deps = extract_js_config_deps(project_root);
    deps.extend(extract_json_config_deps(project_root));
    deps
}
```

```rust
/// Extract workspace glob patterns from project config
fn get_workspace_globs(project_root: &Path) -> Result<Vec<String>, String> {
    // Try pnpm-workspace.yaml first
    let pnpm_ws = project_root.join("pnpm-workspace.yaml");
    if pnpm_ws.exists() {
        let content = std::fs::read_to_string(&pnpm_ws)
            .map_err(|e| format!("Failed to read pnpm-workspace.yaml: {e}"))?;
        let yaml: serde_json::Value = serde_yaml::from_str(&content)
            .map_err(|e| format!("Failed to parse pnpm-workspace.yaml: {e}"))?;

        if let Some(packages) = yaml.get("packages").and_then(|p| p.as_array()) {
            let globs: Vec<String> = packages
                .iter()
                .filter_map(|v| v.as_str().map(String::from))
                .collect();
            return Ok(globs);
        }
    }

    // Try package.json workspaces field
    let pkg_json = project_root.join("package.json");
    if pkg_json.exists() {
        let content = std::fs::read_to_string(&pkg_json)
            .map_err(|e| format!("Failed to read package.json: {e}"))?;
        let json: serde_json::Value = serde_json::from_str(&content)
            .map_err(|e| format!("Failed to parse package.json: {e}"))?;

        if let Some(workspaces) = json.get("workspaces") {
            // Array format: ["packages/*", "apps/*"]
            if let Some(arr) = workspaces.as_array() {
                let globs: Vec<String> = arr
                    .iter()
                    .filter_map(|v| v.as_str().map(String::from))
                    .collect();
                return Ok(globs);
            }
            // Object format: { "packages": ["packages/*"] }
            if let Some(obj) = workspaces.as_object() {
                if let Some(pkgs) = obj.get("packages").and_then(|p| p.as_array())
                {
                    let globs: Vec<String> = pkgs
                        .iter()
                        .filter_map(|v| v.as_str().map(String::from))
                        .collect();
                    return Ok(globs);
                }
            }
        }
    }

    Err("No workspace configuration found".to_string())
}
```

**Step 4: 实现 glob 展开** — 继续在 `src/workspace/mod.rs` 中追加：

```rust
/// Detect workspace packages, returns list of package directory paths.
/// Each returned path contains a package.json.
pub fn detect_workspaces(project_root: &Path) -> Result<Vec<PathBuf>, String> {
    let globs = get_workspace_globs(project_root)?;
    let mut packages = Vec::new();

    for pattern in &globs {
        // Convert workspace glob to filesystem glob
        // "packages/*" → "{root}/packages/*/package.json"
        let full_pattern = format!(
            "{}/{}/package.json",
            project_root.display(),
            pattern.trim_end_matches('/')
        );

        let entries = glob(&full_pattern)
            .map_err(|e| format!("Invalid glob pattern '{pattern}': {e}"))?;

        for entry in entries {
            if let Ok(pkg_json_path) = entry {
                if let Some(pkg_dir) = pkg_json_path.parent() {
                    packages.push(pkg_dir.to_path_buf());
                }
            }
        }
    }

    // Sort for stable output
    packages.sort();
    packages.dedup();

    Ok(packages)
}
```

### Task 8: monorepo 集成到 main.rs

**Files:** `src/main.rs`, `src/i18n.rs`

**背景:** 当检测到 monorepo 时，需要遍历每个 workspace 包独立执行分析（读取子包 package.json 依赖 → 扫描子包源码/配置/scripts → 对比 unused），lockfile 解析只在根目录做一次（共享）。报告按 workspace 包分组显示。

**Step 1: 添加辅助函数** — 在 `src/main.rs` 中提取单包分析逻辑为独立函数，方便 monorepo 复用：

```rust
/// Read package.json dependencies and devDependencies as a DependencyGraph subset
fn read_package_deps(pkg_dir: &Path) -> Option<(
    std::collections::HashMap<String, lockpick::PackageInfo>,
    std::collections::HashMap<String, lockpick::PackageInfo>,
)> {
    let pkg_path = pkg_dir.join("package.json");
    let content = std::fs::read_to_string(&pkg_path).ok()?;
    let json: serde_json::Value = serde_json::from_str(&content).ok()?;

    let mut deps = std::collections::HashMap::new();
    let mut dev_deps = std::collections::HashMap::new();

    if let Some(obj) = json.get("dependencies").and_then(|d| d.as_object()) {
        for (name, version) in obj {
            deps.insert(
                name.clone(),
                lockpick::PackageInfo {
                    name: name.clone(),
                    version: version.as_str().unwrap_or("*").to_string(),
                    integrity: None,
                },
            );
        }
    }

    if let Some(obj) = json.get("devDependencies").and_then(|d| d.as_object()) {
        for (name, version) in obj {
            dev_deps.insert(
                name.clone(),
                lockpick::PackageInfo {
                    name: name.clone(),
                    version: version.as_str().unwrap_or("*").to_string(),
                    integrity: None,
                },
            );
        }
    }

    Some((deps, dev_deps))
}
```

```rust
    if lockpick::workspace::is_monorepo(&project_path) {
        let workspace_packages = match lockpick::workspace::detect_workspaces(&project_path) {
            Ok(pkgs) => pkgs,
            Err(e) => {
                eprintln!("Error detecting workspaces: {e}");
                std::process::exit(1);
            }
        };

        if cli.verbose {
            eprintln!(
                "{} {} {}",
                i18n.t("monorepo_detected"),
                workspace_packages.len(),
                i18n.t("workspace_packages")
            );
        }

        let mut all_has_unused = false;

        for pkg_dir in &workspace_packages {
            let pkg_name = pkg_dir
                .file_name()
                .map(|n| n.to_string_lossy().to_string())
                .unwrap_or_else(|| "unknown".to_string());

            eprintln!("\n--- {} {} ---", i18n.t("workspace_package"), pkg_name);

            // Read this package's deps from its package.json
            let (pkg_deps, pkg_dev_deps) = match read_package_deps(pkg_dir) {
                Some(d) => d,
                None => {
                    eprintln!("  {}", i18n.t("skip_no_deps"));
                    continue;
                }
            };

            // Build a scoped DependencyGraph for this package
            let pkg_graph = lockpick::DependencyGraph {
                dependencies: pkg_deps,
                dev_dependencies: pkg_dev_deps,
                lockfile_type: graph.lockfile_type.clone(),
                all_packages: graph.all_packages.clone(),
            };
```
